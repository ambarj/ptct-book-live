Coding Euler's Method

You know the formula: \( x_{n+1} = x_n + h \cdot f(x_n, t_n) \). You know how loops work. Now it’s time to assemble the pieces into a complete, working solver — one that can tackle any first-order ODE you throw at it.

Learning Objectives
  • Assemble a complete Euler solver from initialization, loop, and storage
  • Solve any first-order ODE \( dx/dt = f(x, t) \) by changing one line of code
  • Explore how initial conditions, step size, and the ODE itself shape the solution

The Three Phases:

1. Initialize: set \(t_0\), \(x_0\), \(h\), \(N\); create storage arrays
2. Loop: repeat the update rule \(N\) times
3. Plot: visualize the result

Python: Your First Euler Solver

Python 

import numpy as np
import plotly.graph_objects as go

# ---- PHASE 1: Initialize ----
def f(x, t):
    return -2 * x          # The ODE: dx/dt = -2x

x0 = 1.0                   # Initial condition
t_max = 3.0                # Solve from t=0 to t=3
h = 0.2                    # Step size
N = int(t_max / h)         # Number of steps

# Storage arrays
t_list = [0.0]
x_list = [x0]

# ---- PHASE 2: Loop ----
t, x = 0.0, x0
for i in range(N):
    x = x + h * f(x, t)    # Euler update
    t = t + h               # Advance time
    t_list.append(t)
    x_list.append(x)

# ---- PHASE 3: Plot ----
t_exact = np.linspace(0, t_max, 300)
x_exact = np.exp(-2 * t_exact)

fig = go.Figure()
fig.add_trace(go.Scatter(x=t_exact, y=x_exact, name='Exact',
    line=dict(color='#808080', width=2, dash='dash')))
fig.add_trace(go.Scatter(x=t_list, y=x_list, name=f'Euler (h={h})',
    line=dict(color='#00f3ff', width=2), mode='lines+markers',
    marker=dict(size=5)))

fig.update_layout(
    template='plotly_dark',
    title='Euler Solver: dx/dt = -2x',
    xaxis=dict(title='t', range=[0, t_max],
        gridcolor='#2a2f4a', zerolinecolor='#808080', linecolor='#808080'),
    yaxis=dict(title='x(t)', range=[-0.2, 1.2],
        gridcolor='#2a2f4a', zerolinecolor='#808080', linecolor='#808080'),
)
fig.show()

print(f"Steps: {N}")
print(f"Final Euler: x({t_max}) = {x_list[-1]:.6f}")
print(f"Exact:       x({t_max}) = {np.exp(-2*t_max):.6f}")
print(f"Error:       {abs(x_list[-1] - np.exp(-2*t_max)):.6f}")
Python 

import numpy as np
import plotly.graph_objects as go

def euler_solve(f, x0, t_max, h):
    """Reusable Euler solver — change f to solve any ODE."""
    t_list, x_list = [0.0], [x0]
    t, x = 0.0, x0
    for i in range(int(t_max / h)):
        x = x + h * f(x, t)
        t = t + h
        t_list.append(t)
        x_list.append(x)
    return np.array(t_list), np.array(x_list)

# Three different ODEs, same solver
odes = {
    'Decay':    (lambda x, t: -2*x,        1.0),
    'Logistic': (lambda x, t: x*(1-x),     0.1),
    'Driven':   (lambda x, t: -x + np.sin(t), 0.0),
}

colors = {'Decay': '#00f3ff', 'Logistic': '#ff00ff', 'Driven': '#ffff00'}

fig = go.Figure()
for name, (f, x0) in odes.items():
    t, x = euler_solve(f, x0, t_max=6.0, h=0.05)
    fig.add_trace(go.Scatter(x=t, y=x, name=name,
        line=dict(color=colors[name], width=2)))

fig.update_layout(
    template='plotly_dark',
    title='One Solver, Three ODEs',
    xaxis=dict(title='t', range=[0, 6],
        gridcolor='#2a2f4a', zerolinecolor='#808080', linecolor='#808080'),
    yaxis=dict(title='x(t)', range=[-0.5, 1.5],
        gridcolor='#2a2f4a', zerolinecolor='#808080', linecolor='#808080'),
)
fig.show()
